📏 SIZE REFERENCE:
   icon-xs  = 16px  (inline text)
   icon-sm  = 20px  (buttons)
   icon-md  = 24px  (default)
   icon-lg  = 32px  (headers)
   icon-xl  = 48px  (features)
   icon-2xl = 64px  (success states)
   icon-3xl = 96px  (major states) ← Used in signup flow

   text-danger  = Red (#DC2626)
   text-mint    = Mint Green (#589D96)
